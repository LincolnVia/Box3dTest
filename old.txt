
void World::ApplyCompanionCubeReleasedTuning() {
  for (auto &object : sceneObjects) {
    if (object.name != "companion_cube") {
      continue;
    }
    if (!b3Body_IsEnabled(object.bodyId)) {
      continue;
    }

    // Held cubes have gravity disabled by Player, so leave them fully under the
    // pickup joint. Ground drag only affects released cubes.
    if (b3Body_GetGravityScale(object.bodyId) <= 0.0f) {
      continue;
    }

    // Player movement is velocity-driven, so a released cube can gain too much
    // solver energy if it is crushed against a wall. Clamp the free cube every
    // step, not just when it is dropped.
    ClampBodyVelocity(object.bodyId, CompanionCubeStats::freeMaxSpeed,
                      CompanionCubeStats::freeMaxAngularSpeed);
    ResolveStaticPenetration(object.bodyId, object.shapeId,
                             CompanionCubeStats::staticPenetrationSlop,
                             CompanionCubeStats::staticDepenetrationBias);

    b3QueryFilter filter = b3DefaultQueryFilter();
    filter.maskBits = CollisionCategory::Static;

    const b3RayResult groundHit = b3World_CastRayClosest(
        worldId, b3Body_GetPosition(object.bodyId),
        {0.0f, -object.halfExtents.y - CompanionCubeStats::groundCheckDistance,
         0.0f},
        filter);

    if (!groundHit.hit) {
      continue;
    }

    const b3Vec3 velocity = b3Body_GetLinearVelocity(object.bodyId);
    const b3Vec3 horizontalVelocity = {velocity.x, 0.0f, velocity.z};
    ApplyVectorDrag(object.bodyId, horizontalVelocity,
                    CompanionCubeStats::groundLinearDrag,
                    CompanionCubeStats::groundStopSpeed, timeStep, true);

    ApplyVectorDrag(object.bodyId, b3Body_GetAngularVelocity(object.bodyId),
                    CompanionCubeStats::groundAngularDrag,
                    CompanionCubeStats::groundStopAngularSpeed, timeStep,
                    false);
  }
}
#include <iostream>
#include "EventListener.h" // Assuming code is in this header

// 1. A simple standalone global function callback
void globalLogger(size_t id, bool state, void* userData) {
    std::cout << "[Global Log] System ID " << id 
              << " changed to " << (state ? "ON" : "OFF") << "\n";
}

// 2. An object-oriented subscriber example
class AudioSystem {
public:
    void playToggleSound(size_t id, bool state) {
        std::cout << "[Audio System] Playing sound for ID " << id 
                  << " turning " << (state ? "UP" : "DOWN") << "\n";
    }
};

// Wrapper callback to bridge the C-style function pointer to a C++ class instance
void audioCallbackBridge(size_t id, bool state, void* userData) {
    // Cast the generic void pointer back to our specific class instance
    AudioSystem* audioInstance = static_cast<AudioSystem*>(userData);
    if (audioInstance) {
        audioInstance->playToggleSound(id, state);
    }
}

int main() {
    EventListener listener;
    AudioSystem audio;

    // Register Subscriber 1: The standalone global logger
    listener.subscribe(globalLogger);

    // Register Subscriber 2: The AudioSystem object (passing its instance address as context)
    listener.subscribe(audioCallbackBridge, &audio);

    std::cout << "--- Triggering ID 12 (Within 64 range) ---\n";
    listener.setState(12, true);  // Triggers BOTH subscribers
    listener.setState(12, true);  // No change, ignored completely

    std::cout << "\n--- Triggering ID 105 (Exceeds original 64 limit) ---\n";
    // The vector seamlessly scales up to hold 105 bits on the fly
    listener.setState(105, true); // Triggers BOTH subscribers

    std::cout << "\n--- Flipping ID 12 back to false ---\n";
    listener.setState(12, false); // Triggers BOTH subscribers

    return 0;
}
